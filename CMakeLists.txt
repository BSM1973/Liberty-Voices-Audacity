cmake_minimum_required(VERSION 3.22)
project(VoicesAudacity VERSION 0.8.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
option(VOICES_DSP_ONLY "Build portable DSP tests without JUCE" OFF)
enable_testing()
add_executable(VoicesDSPTests Tests/DspTests.cpp)
add_test(NAME VoicesDSPTests COMMAND VoicesDSPTests)
add_executable(VoicesMasteringTests Tests/MasteringTests.cpp)
add_test(NAME VoicesMasteringTests COMMAND VoicesMasteringTests)
add_executable(VoicesPrecisionTests Tests/PrecisionTests.cpp)
add_test(NAME VoicesPrecisionTests COMMAND VoicesPrecisionTests)
if(VOICES_DSP_ONLY)
  return()
endif()
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
set(formats VST3 Standalone)
if(APPLE)
  list(APPEND formats AU)
endif()
juce_add_plugin(VoicesAudacity COMPANY_NAME "Liberty Music Software" BUNDLE_ID com.libertymusicsoftware.voicesaudacity PLUGIN_MANUFACTURER_CODE Lbty PLUGIN_CODE Voau FORMATS ${formats} PRODUCT_NAME "Voices Audacity" IS_SYNTH FALSE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE AU_MAIN_TYPE kAudioUnitType_Effect VST3_CATEGORIES Fx Pitch Shift)
target_sources(VoicesAudacity PRIVATE Source/Plugin.cpp Source/Editor.cpp)
target_compile_definitions(VoicesAudacity PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
if(MSVC)
 target_compile_options(VoicesAudacity PRIVATE /utf-8)
endif()
target_link_libraries(VoicesAudacity PRIVATE juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
